#cmake_minimum_required(VERSION 3.13)

# Robo Pico uses a Pico W. Must be set BEFORE the SDK import.
set(PICO_BOARD pico_w)

# Copy pico_sdk_import.cmake from <pico-sdk>/external/ into this folder.
include(pico_sdk_import.cmake)

project(buddy3_barcode_bringup C CXX ASM)
set(CMAKE_C_STANDARD 11)

pico_sdk_init()

# Pure-logic files (no RTOS, no hardware): the same ones the real tasks use.
set(BUDDY3_LOGIC
    bar_capture.c
    barcode_decode.c
    line_logic.c          # not used by the bring-up yet; harmless
)

add_executable(barcode_bringup
    barcode_bringup.c
    ${BUDDY3_LOGIC}
)

target_include_directories(barcode_bringup PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(barcode_bringup pico_stdlib)      # gpio + timer + stdio

pico_enable_stdio_usb(barcode_bringup 1)    # printf over USB serial
pico_enable_stdio_uart(barcode_bringup 0)

pico_add_extra_outputs(barcode_bringup)     # makes the .uf2 to drag onto the Pico
